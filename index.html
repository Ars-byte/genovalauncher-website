Get Started

Install in
seconds.

1

Download the latest release

bash
$wget https://github.com/Ars-byte/PYmc-launcher/releases/latest/download/PYmc-v1.0.0.tar.gz
2

Extract and run

bash
$tar xzf PYmc-v1.0.0.tar.gz
$cd PYmc-v1.0.0
$./PYmc.sh
3

Or build from source

bash
$git clone https://github.com/Ars-byte/PYmc-launcher.git
$cd PYmc-launcher && ./run.sh
1

Add PYmc to flake.nix

flake.nix
inputs = {
  pymc.url = "github:Ars-byte/PYmc-launcher";
};
2

Add to system packages

configuration.nix
environment.systemPackages = [
  inputs.pymc.packages.${system}.default
];
3

Rebuild and run

bash
$sudo nixos-rebuild switch
$pymc
